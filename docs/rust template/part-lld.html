#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum VehicleType {
    Motorcycle,
    Car,
    Bus,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum SpotSize {
    Compact,
    Regular,
    Large,
}

/// Capability check lives on a trait so sizing policy can be swapped/mocked.
trait FitsIn {
    fn fits_in(&self, size: SpotSize) -> bool;
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Vehicle {
    plate: String,
    kind: VehicleType,
}

impl FitsIn for Vehicle {
    fn fits_in(&self, size: SpotSize) -> bool {
        match (size, self.kind) {
            (SpotSize::Compact, VehicleType::Motorcycle) => true,
            (SpotSize::Regular, VehicleType::Motorcycle | VehicleType::Car) => true,
            (SpotSize::Large, _) => true,
            _ => false,
        }
    }
}

#[derive(Debug)]
struct Spot {
    size: SpotSize,
    vehicle: Option<Vehicle>,
}

impl Spot {
    fn new(size: SpotSize) -> Self {
        Self {
            size,
            vehicle: None,
        }
    }

    fn is_free(&self) -> bool {
        self.vehicle.is_none()
    }

    fn can_park(&self, vehicle: &Vehicle) -> bool {
        self.is_free() && vehicle.fits_in(self.size)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct SpotId {
    floor: usize,
    index: usize,
}

#[derive(Debug)]
struct Floor {
    spots: Vec<Spot>,
}

#[derive(Debug)]
struct ParkingLot {
    floors: Vec<Floor>,
}

#[derive(Debug, PartialEq, Eq)]
enum ParkError {
    Full,
    AlreadyParked,
}

#[derive(Debug, PartialEq, Eq)]
enum LeaveError {
    NotFound,
}

impl ParkingLot {
    fn new(floors: Vec<Vec<SpotSize>>) -> Self {
        Self {
            floors: floors
                .into_iter()
                .map(|sizes| Floor {
                    spots: sizes.into_iter().map(Spot::new).collect(),
                })
                .collect(),
        }
    }

    fn find_vehicle(&self, plate: &str) -> Option<SpotId> {
        for (floor, f) in self.floors.iter().enumerate() {
            for (index, spot) in f.spots.iter().enumerate() {
                if spot
                    .vehicle
                    .as_ref()
                    .map(|v| v.plate == plate)
                    .unwrap_or(false)
                {
                    return Some(SpotId { floor, index });
                }
            }
        }
        None
    }

    /// Nearest free compatible spot: lowest floor, then lowest index.
    fn find_nearest(&self, vehicle: &Vehicle) -> Option<SpotId> {
        for (floor, f) in self.floors.iter().enumerate() {
            for (index, spot) in f.spots.iter().enumerate() {
                if spot.can_park(vehicle) {
                    return Some(SpotId { floor, index });
                }
            }
        }
        None
    }

    fn park(&mut self, vehicle: Vehicle) -> Result<SpotId, ParkError> {
        if self.find_vehicle(&vehicle.plate).is_some() {
            return Err(ParkError::AlreadyParked);
        }
        let id = self.find_nearest(&vehicle).ok_or(ParkError::Full)?;
        self.floors[id.floor].spots[id.index].vehicle = Some(vehicle);
        Ok(id)
    }

    fn leave(&mut self, plate: &str) -> Result<Vehicle, LeaveError> {
        let id = self.find_vehicle(plate).ok_or(LeaveError::NotFound)?;
        Ok(self.floors[id.floor].spots[id.index]
            .vehicle
            .take()
            .expect("find_vehicle guarantees occupied"))
    }
}

fn main() {
    let mut lot = ParkingLot::new(vec![
        vec![SpotSize::Compact, SpotSize::Regular],
        vec![SpotSize::Large],
    ]);

    let car = Vehicle {
        plate: "CAR-1".into(),
        kind: VehicleType::Car,
    };
    let id = lot.park(car).unwrap();
    assert_eq!(id, SpotId { floor: 0, index: 1 });

    let bike = Vehicle {
        plate: "BIKE-1".into(),
        kind: VehicleType::Motorcycle,
    };
    let id2 = lot.park(bike).unwrap();
    assert_eq!(id2, SpotId { floor: 0, index: 0 });

    let bus = Vehicle {
        plate: "BUS-1".into(),
        kind: VehicleType::Bus,
    };
    let id3 = lot.park(bus).unwrap();
    assert_eq!(id3, SpotId { floor: 1, index: 0 });

    assert_eq!(
        lot.park(Vehicle {
            plate: "X".into(),
            kind: VehicleType::Car,
        }),
        Err(ParkError::Full)
    );

    let left = lot.leave("CAR-1").unwrap();
    assert_eq!(left.plate, "CAR-1");
    assert_eq!(lot.leave("NOPE"), Err(LeaveError::NotFound));

    let again = lot
        .park(Vehicle {
            plate: "CAR-2".into(),
            kind: VehicleType::Car,
        })
        .unwrap();
    assert_eq!(again, SpotId { floor: 0, index: 1 });
}

  • ★★★Uses enums + a FitsIn trait so vehicle/spot compatibility is data-driven and swappable, not a tangle of if kind == … in park.
  • ★★Nearest-spot scan is deterministic (floor then index) and park/leave return typed errors (Full / AlreadyParked / NotFound) instead of panicking.
  • ★Separates spot identity from occupancy — e.g. SpotId { floor, index } — so leave/find stay clear at the API boundary.

  • P1"Why a trait instead of a method on VehicleType?" — Traits let you mock sizing policy in tests and keep park generic over anything that FitsIn; an enum method is fine too if justified.
  • P2"How do you prevent the same plate parking twice?" — Scan for an existing plate before allocate (or maintain a plate→spot index). Return AlreadyParked.
  • P3"What changes for concurrent park from many threads?" — A lot mutex, per-floor locks, or sharded free-lists; discuss the race between find_nearest and claim.

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Direction {
    Up,
    Down,
    Idle,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum ElevatorState {
    Idle,
    Moving(Direction),
    DoorOpen,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct Request {
    floor: u32,
    /// Desired travel after boarding; `None` means hall call / go-to-floor only.
    intent: Option<Direction>,
}

#[derive(Debug)]
struct Elevator {
    current: u32,
    state: ElevatorState,
    /// Pending stops, kept sorted ascending for the scan policy.
    stops: Vec<u32>,
    /// Direction we are currently serving under SCAN (even while doors open).
    serving: Direction,
}

impl Elevator {
    fn new(start: u32) -> Self {
        Self {
            current: start,
            state: ElevatorState::Idle,
            stops: Vec::new(),
            serving: Direction::Idle,
        }
    }

    fn request(&mut self, req: Request) {
        if !self.stops.contains(&req.floor) {
            self.stops.push(req.floor);
            self.stops.sort_unstable();
        }
        if self.serving == Direction::Idle {
            self.serving = if req.floor > self.current {
                Direction::Up
            } else if req.floor < self.current {
                Direction::Down
            } else {
                // Same-floor: intent chooses departure direction after boarding.
                req.intent.unwrap_or(Direction::Idle)
            };
            if self.serving != Direction::Idle {
                self.state = ElevatorState::Moving(self.serving);
            } else if req.floor == self.current {
                self.state = ElevatorState::DoorOpen;
            }
        } else if req.floor == self.current {
            // Same-floor while busy: open doors. Intent seeds serving only if
            // we somehow lost direction; otherwise SCAN keeps the sweep.
            self.serving = match req.intent {
                Some(dir) if self.serving == Direction::Idle => dir,
                _ => self.serving,
            };
            self.state = ElevatorState::DoorOpen;
        }
    }

    /// One discrete step: move one floor toward the next stop, or open/close doors.
    fn step(&mut self) {
        match self.state {
            ElevatorState::DoorOpen => {
                self.stops.retain(|f| *f != self.current);
                self.pick_next_direction();
                if self.serving == Direction::Idle {
                    self.state = ElevatorState::Idle;
                } else {
                    self.state = ElevatorState::Moving(self.serving);
                }
            }
            ElevatorState::Idle => {
                if let Some(&next) = self.next_stop() {
                    self.serving = if next > self.current {
                        Direction::Up
                    } else if next < self.current {
                        Direction::Down
                    } else {
                        Direction::Idle
                    };
                    if next == self.current {
                        self.state = ElevatorState::DoorOpen;
                    } else {
                        self.state = ElevatorState::Moving(self.serving);
                    }
                }
            }
            ElevatorState::Moving(dir) => {
                match dir {
                    Direction::Up => self.current += 1,
                    Direction::Down => self.current = self.current.saturating_sub(1),
                    Direction::Idle => {}
                }
                if self.stops.contains(&self.current) {
                    self.state = ElevatorState::DoorOpen;
                }
            }
        }
    }

    fn next_stop(&self) -> Option<&u32> {
        match self.serving {
            Direction::Up => self.stops.iter().find(|&&f| f >= self.current),
            Direction::Down => self.stops.iter().rev().find(|&&f| f <= self.current),
            Direction::Idle => self.stops.first(),
        }
    }

    fn pick_next_direction(&mut self) {
        if self.stops.is_empty() {
            self.serving = Direction::Idle;
            return;
        }
        match self.serving {
            Direction::Up => {
                if self.stops.iter().any(|&f| f > self.current) {
                    self.serving = Direction::Up;
                } else if self.stops.iter().any(|&f| f < self.current) {
                    self.serving = Direction::Down;
                } else {
                    self.serving = Direction::Idle;
                }
            }
            Direction::Down => {
                if self.stops.iter().any(|&f| f < self.current) {
                    self.serving = Direction::Down;
                } else if self.stops.iter().any(|&f| f > self.current) {
                    self.serving = Direction::Up;
                } else {
                    self.serving = Direction::Idle;
                }
            }
            Direction::Idle => {
                let Some(&next) = self.stops.first() else {
                    return;
                };
                self.serving = if next > self.current {
                    Direction::Up
                } else if next < self.current {
                    Direction::Down
                } else {
                    Direction::Idle
                };
            }
        }
    }

    fn run_until_idle(&mut self, max_steps: usize) {
        for _ in 0..max_steps {
            if self.state == ElevatorState::Idle && self.stops.is_empty() {
                return;
            }
            self.step();
        }
    }
}

fn main() {
    let mut e = Elevator::new(0);
    e.request(Request {
        floor: 3,
        intent: Some(Direction::Up),
    });
    e.request(Request {
        floor: 1,
        intent: Some(Direction::Up),
    });
    e.request(Request {
        floor: 5,
        intent: None,
    });

    // SCAN upward first: visit 1, then 3, then 5.
    let mut visited = Vec::new();
    for _ in 0..40 {
        if e.state == ElevatorState::DoorOpen {
            visited.push(e.current);
        }
        if e.state == ElevatorState::Idle && e.stops.is_empty() {
            break;
        }
        e.step();
    }
    assert_eq!(visited, vec![1, 3, 5]);
    assert_eq!(e.current, 5);
    assert_eq!(e.state, ElevatorState::Idle);

    // Now reverse: from 5, serve down requests.
    e.request(Request {
        floor: 2,
        intent: Some(Direction::Down),
    });
    e.request(Request {
        floor: 4,
        intent: Some(Direction::Down),
    });
    visited.clear();
    for _ in 0..40 {
        if e.state == ElevatorState::DoorOpen {
            visited.push(e.current);
        }
        if e.state == ElevatorState::Idle && e.stops.is_empty() {
            break;
        }
        e.step();
    }
    assert_eq!(visited, vec![4, 2]);
    assert_eq!(e.state, ElevatorState::Idle);

    e.run_until_idle(5);
    assert_eq!(e.state, ElevatorState::Idle);
}
What to look for — unprompted signals
  • ★★★Explicit state enum + serving direction — movement, door open, and idle are transitions, not boolean flags; SCAN keeps a serving direction across door cycles.
  • ★★Implements SCAN correctly — collects stops, prefers continuing up/down, reverses only when no further stop in that direction; visit order asserted in main.
  • ★Uses request intent for same-floor / idle bootstrapping without breaking an in-progress sweep — shows hall-call semantics.
Follow-up probes
  • P1"SCAN vs LOOK vs FCFS — when does it matter?" — FCFS starves under load; SCAN/LOOK bound travel like a disk arm. LOOK stops at the last request instead of the shaft end.
  • P2"How would you extend this to N elevators?" — Dispatcher assigns a request to the car minimizing cost (distance, direction alignment, load); each car still runs its own SCAN state machine.
  • P3"Where do you put time — doors open for N steps?" — Count door ticks in the state machine or inject a clock; keep step as the single advance point for tests.
🎯 Senior angle / Rust angle
Elevator interviews reward a crisp state machine more than clever heaps. In Rust, encoding ElevatorState and Direction as enums makes illegal combinations unrepresentable; SCAN is then “pick next stop in serving, else flip” — easy to assert in main.

use std::time::Duration;

/// Token-bucket rate limiter as a small library API.
/// Time is injected via a clock callback so tests stay deterministic.
struct TokenBucket {
    capacity: f64,
    /// Tokens added per second.
    refill_per_sec: f64,
    tokens: f64,
    /// Monotonic time in milliseconds since an arbitrary epoch.
    last_ms: u64,
}

#[derive(Debug, PartialEq, Eq)]
enum LimitError {
    /// Requested more tokens than the bucket can ever hold.
    ExceedsCapacity,
}

impl TokenBucket {
    fn new(capacity: f64, refill_per_sec: f64, now_ms: u64) -> Self {
        assert!(capacity > 0.0 && refill_per_sec >= 0.0);
        Self {
            capacity,
            refill_per_sec,
            tokens: capacity,
            last_ms: now_ms,
        }
    }

    fn refill(&mut self, now_ms: u64) {
        if now_ms > self.last_ms {
            let elapsed_secs = (now_ms - self.last_ms) as f64 / 1000.0;
            self.tokens = (self.tokens + elapsed_secs * self.refill_per_sec).min(self.capacity);
            self.last_ms = now_ms;
        }
    }

    /// Try to consume `n` tokens at `now_ms`. Returns Ok(true) if allowed,
    /// Ok(false) if rate-limited, Err if `n` can never succeed.
    fn try_acquire(&mut self, n: f64, now_ms: u64) -> Result<bool, LimitError> {
        if n > self.capacity {
            return Err(LimitError::ExceedsCapacity);
        }
        self.refill(now_ms);
        if self.tokens >= n {
            self.tokens -= n;
            Ok(true)
        } else {
            Ok(false)
        }
    }

    fn available(&mut self, now_ms: u64) -> f64 {
        self.refill(now_ms);
        self.tokens
    }

    /// Convenience: wait duration until `n` tokens would be available (approx).
    fn time_until(&mut self, n: f64, now_ms: u64) -> Result<Duration, LimitError> {
        if n > self.capacity {
            return Err(LimitError::ExceedsCapacity);
        }
        self.refill(now_ms);
        if self.tokens >= n {
            return Ok(Duration::ZERO);
        }
        if self.refill_per_sec == 0.0 {
            return Ok(Duration::from_secs(u64::MAX / 2));
        }
        let need = n - self.tokens;
        let secs = need / self.refill_per_sec;
        Ok(Duration::from_secs_f64(secs))
    }
}

fn main() {
    let mut bucket = TokenBucket::new(5.0, 1.0, 0);
    assert!(bucket.try_acquire(5.0, 0).unwrap());
    assert!(!bucket.try_acquire(1.0, 0).unwrap());
    assert_eq!(bucket.available(0), 0.0);

    // After 2 seconds, 2 tokens refilled.
    assert!(bucket.try_acquire(2.0, 2_000).unwrap());
    assert!(!bucket.try_acquire(1.0, 2_000).unwrap());

    assert_eq!(
        bucket.try_acquire(10.0, 2_000),
        Err(LimitError::ExceedsCapacity)
    );

    let wait = bucket.time_until(1.0, 2_000).unwrap();
    assert!(wait >= Duration::from_secs(1));

    // Burst capacity restored after long idle.
    assert!((bucket.available(10_000) - 5.0).abs() < 1e-9);
    assert!(bucket.try_acquire(5.0, 10_000).unwrap());
}
What to look for — unprompted signals
  • ★★★Injects now_ms (no hidden Instant::now()) so refill, deny, and time_until are assertable without sleeps.
  • ★★Distinguishes rate-limit deny from exceeds-capacity — Ok(false) vs Err(ExceedsCapacity) — a library-quality API detail.
  • ★Refill is lazy on access and caps at capacity; can explain float tokens for a demo vs integer millis math in production.
Follow-up probes
  • P1"Why not sleep inside try_acquire?" — Libraries should not block the caller’s scheduler. Return false / time_until and let the app wait or shed load.
  • P2"How does this map to a distributed limiter?" — Local token bucket ≈ single-node; Redis INCR+TTL or a central counter for multi-node. Discuss drift and burst fairness.
  • P3"Token bucket vs sliding window?" — Bucket allows controlled bursts up to capacity; sliding window smooths per-window counts. Pick based on UX vs abuse.
🎯 Senior angle / Rust angle
Library LLD is API design under a microscope: pure functions of (state, now), typed errors, and no policy hidden in sleeps. Rust interviewers like seeing time as a parameter — the same habit that makes production limiters testable and embeddable in async services.

use std::io::{self, Write};

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
enum Level {
    Error = 1,
    Warn = 2,
    Info = 3,
    Debug = 4,
    Trace = 5,
}

trait Sink {
    fn write_line(&mut self, level: Level, msg: &str) -> io::Result<()>;
    fn flush(&mut self) -> io::Result<()>;
}

struct StderrSink;

impl Sink for StderrSink {
    fn write_line(&mut self, level: Level, msg: &str) -> io::Result<()> {
        let mut err = io::stderr().lock();
        writeln!(err, "[{level:?}] {msg}")
    }

    fn flush(&mut self) -> io::Result<()> {
        io::stderr().flush()
    }
}

/// In-memory sink for tests / capturing.
struct MemSink {
    lines: Vec<(Level, String)>,
}

impl MemSink {
    fn new() -> Self {
        Self { lines: Vec::new() }
    }
}

impl Sink for MemSink {
    fn write_line(&mut self, level: Level, msg: &str) -> io::Result<()> {
        self.lines.push((level, msg.to_string()));
        Ok(())
    }

    fn flush(&mut self) -> io::Result<()> {
        Ok(())
    }
}

/// Buffered logger: hot path appends; I/O happens on flush (async-friendly boundary).
struct Logger<S: Sink> {
    min_level: Level,
    sink: Option<S>,
    buffer: Vec<(Level, String)>,
    capacity: usize,
}

impl<S: Sink> Logger<S> {
    fn new(min_level: Level, sink: S, capacity: usize) -> Self {
        Self {
            min_level,
            sink: Some(sink),
            buffer: Vec::with_capacity(capacity),
            capacity: capacity.max(1),
        }
    }

    fn log(&mut self, level: Level, msg: impl Into<String>) -> io::Result<()> {
        if level > self.min_level {
            return Ok(());
        }
        self.buffer.push((level, msg.into()));
        if self.buffer.len() >= self.capacity {
            self.flush()?;
        }
        Ok(())
    }

    fn info(&mut self, msg: impl Into<String>) -> io::Result<()> {
        self.log(Level::Info, msg)
    }

    fn debug(&mut self, msg: impl Into<String>) -> io::Result<()> {
        self.log(Level::Debug, msg)
    }

    fn error(&mut self, msg: impl Into<String>) -> io::Result<()> {
        self.log(Level::Error, msg)
    }

    fn flush(&mut self) -> io::Result<()> {
        if let Some(sink) = self.sink.as_mut() {
            for (level, msg) in self.buffer.drain(..) {
                sink.write_line(level, &msg)?;
            }
            sink.flush()?;
        }
        Ok(())
    }

    fn into_sink(mut self) -> io::Result<S> {
        self.flush()?;
        Ok(self.sink.take().expect("sink present until into_sink"))
    }

    fn buffered_len(&self) -> usize {
        self.buffer.len()
    }
}

impl<S: Sink> Drop for Logger<S> {
    fn drop(&mut self) {
        let _ = self.flush();
    }
}

fn main() {
    let mut log = Logger::new(Level::Info, MemSink::new(), 2);
    log.info("a").unwrap();
    assert_eq!(log.buffered_len(), 1);

    // Debug filtered out by min_level = Info.
    log.debug("secret").unwrap();
    assert_eq!(log.buffered_len(), 1);

    // Second info hits capacity → auto-flush.
    log.info("b").unwrap();
    assert_eq!(log.buffered_len(), 0);

    log.error("boom").unwrap();
    let sink = log.into_sink().unwrap();
    assert_eq!(
        sink.lines,
        vec![
            (Level::Info, "a".into()),
            (Level::Info, "b".into()),
            (Level::Error, "boom".into()),
        ]
    );

    assert!(Level::Error < Level::Warn);
    assert!(Level::Warn < Level::Info);
    assert!(Level::Info < Level::Debug);
    assert!(Level::Debug < Level::Trace);

    // Production sink type must compile; avoid writing in asserts.
    let mut stderr_log = Logger::new(Level::Error, StderrSink, 8);
    stderr_log.debug("filtered").unwrap();
    assert_eq!(stderr_log.buffered_len(), 0);
}

use std::collections::HashMap;
use std::time::{Duration, Instant};

#[derive(Debug, Clone, PartialEq)]
enum SeatState {
    Available,
    Held { user: String, expires_at: Instant },
    Booked { user: String },
}

#[derive(Debug, PartialEq)]
enum BookingError {
    UnknownSeat,
    NotAvailable,
    HoldExpired,
    WrongUser,
    NotHeld,
}

struct TicketSystem {
    seats: HashMap<String, SeatState>,
    hold_ttl: Duration,
}

impl TicketSystem {
    fn new(seat_ids: &[&str], hold_ttl: Duration) -> Self {
        let mut seats = HashMap::new();
        for id in seat_ids {
            seats.insert((*id).to_string(), SeatState::Available);
        }
        Self { seats, hold_ttl }
    }

    fn purge_expired(&mut self, now: Instant) {
        for state in self.seats.values_mut() {
            if let SeatState::Held { expires_at, .. } = state {
                if *expires_at <= now {
                    *state = SeatState::Available;
                }
            }
        }
    }

    fn hold(&mut self, seat: &str, user: &str, now: Instant) -> Result<(), BookingError> {
        self.purge_expired(now);
        let state = self.seats.get_mut(seat).ok_or(BookingError::UnknownSeat)?;
        match state {
            SeatState::Available => {
                *state = SeatState::Held {
                    user: user.to_string(),
                    expires_at: now + self.hold_ttl,
                };
                Ok(())
            }
            _ => Err(BookingError::NotAvailable),
        }
    }

    fn commit(&mut self, seat: &str, user: &str, now: Instant) -> Result<(), BookingError> {
        // Do not purge first — expired holds must surface as HoldExpired.
        let state = self.seats.get_mut(seat).ok_or(BookingError::UnknownSeat)?;
        match state {
            SeatState::Held {
                user: holder,
                expires_at,
            } => {
                if *expires_at <= now {
                    *state = SeatState::Available;
                    return Err(BookingError::HoldExpired);
                }
                if holder != user {
                    return Err(BookingError::WrongUser);
                }
                *state = SeatState::Booked {
                    user: user.to_string(),
                };
                Ok(())
            }
            SeatState::Available => Err(BookingError::NotHeld),
            SeatState::Booked { .. } => Err(BookingError::NotAvailable),
        }
    }

    fn release_hold(&mut self, seat: &str, user: &str, now: Instant) -> Result<(), BookingError> {
        self.purge_expired(now);
        let state = self.seats.get_mut(seat).ok_or(BookingError::UnknownSeat)?;
        match state {
            SeatState::Held { user: holder, .. } if holder == user => {
                *state = SeatState::Available;
                Ok(())
            }
            SeatState::Held { .. } => Err(BookingError::WrongUser),
            SeatState::Available => Err(BookingError::NotHeld),
            SeatState::Booked { .. } => Err(BookingError::NotAvailable),
        }
    }

    fn seat(&self, id: &str) -> Option<&SeatState> {
        self.seats.get(id)
    }
}

fn main() {
    let ttl = Duration::from_secs(30);
    let t0 = Instant::now();
    let mut sys = TicketSystem::new(&["A1", "A2", "B1"], ttl);

    assert!(sys.hold("A1", "alice", t0).is_ok());
    assert!(matches!(
        sys.seat("A1"),
        Some(SeatState::Held { user, .. }) if user == "alice"
    ));

    // Concurrent hold attempt fails while held.
    assert_eq!(sys.hold("A1", "bob", t0), Err(BookingError::NotAvailable));

    // Wrong user cannot commit.
    assert_eq!(
        sys.commit("A1", "bob", t0 + Duration::from_secs(1)),
        Err(BookingError::WrongUser)
    );

    // Alice commits successfully.
    assert!(sys.commit("A1", "alice", t0 + Duration::from_secs(1)).is_ok());
    assert_eq!(
        sys.seat("A1"),
        Some(&SeatState::Booked {
            user: "alice".into()
        })
    );

    // Hold expires before commit.
    assert!(sys.hold("A2", "bob", t0).is_ok());
    assert_eq!(
        sys.commit("A2", "bob", t0 + Duration::from_secs(31)),
        Err(BookingError::HoldExpired)
    );
    assert_eq!(sys.seat("A2"), Some(&SeatState::Available));

    // After expiry, another user can hold and book.
    assert!(sys.hold("A2", "carol", t0 + Duration::from_secs(32)).is_ok());
    assert!(sys
        .commit("A2", "carol", t0 + Duration::from_secs(33))
        .is_ok());

    // Release hold returns seat to pool.
    assert!(sys.hold("B1", "dave", t0).is_ok());
    assert!(sys.release_hold("B1", "dave", t0).is_ok());
    assert_eq!(sys.seat("B1"), Some(&SeatState::Available));

use std::collections::HashMap;

type Listener = Box<dyn Fn(&str)>;

struct SubscriptionId(u64);

struct PubSub {
    next_id: u64,
    topics: HashMap<String, Vec<(u64, Listener)>>,
}

impl PubSub {
    fn new() -> Self {
        Self {
            next_id: 1,
            topics: HashMap::new(),
        }
    }

    fn subscribe<F>(&mut self, topic: &str, listener: F) -> SubscriptionId
    where
        F: Fn(&str) + 'static,
    {
        let id = self.next_id;
        self.next_id += 1;
        self.topics
            .entry(topic.to_string())
            .or_default()
            .push((id, Box::new(listener)));
        SubscriptionId(id)
    }

    fn unsubscribe(&mut self, topic: &str, id: SubscriptionId) -> bool {
        let Some(list) = self.topics.get_mut(topic) else {
            return false;
        };
        let before = list.len();
        list.retain(|(sid, _)| *sid != id.0);
        before != list.len()
    }

    fn publish(&self, topic: &str, message: &str) -> usize {
        match self.topics.get(topic) {
            Some(list) => {
                for (_, listener) in list {
                    listener(message);
                }
                list.len()
            }
            None => 0,
        }
    }
}

fn main() {
    use std::cell::RefCell;
    use std::rc::Rc;

    let mut bus = PubSub::new();
    let inbox_a = Rc::new(RefCell::new(Vec::<String>::new()));
    let inbox_b = Rc::new(RefCell::new(Vec::<String>::new()));

    let a = inbox_a.clone();
    let id_a = bus.subscribe("orders", move |msg| {
        a.borrow_mut().push(msg.to_string());
    });

    let b = inbox_b.clone();
    let id_b = bus.subscribe("orders", move |msg| {
        b.borrow_mut().push(format!("B:{msg}"));
    });

    assert_eq!(bus.publish("orders", "sku-1"), 2);
    assert_eq!(*inbox_a.borrow(), vec!["sku-1".to_string()]);
    assert_eq!(*inbox_b.borrow(), vec!["B:sku-1".to_string()]);

    assert!(bus.unsubscribe("orders", id_b));
    assert_eq!(bus.publish("orders", "sku-2"), 1);
    assert_eq!(
        *inbox_a.borrow(),
        vec!["sku-1".to_string(), "sku-2".to_string()]
    );
    assert_eq!(*inbox_b.borrow(), vec!["B:sku-1".to_string()]);

    assert!(bus.unsubscribe("orders", id_a));
    assert_eq!(bus.publish("orders", "sku-3"), 0);
    assert_eq!(bus.publish("unknown", "x"), 0);
}

  • ★★Uses Fn (not only FnMut) or documents interior mutability — publish(&self) can call listeners; demo captures use Rc<RefCell<Vec>> to record messages.

  • P2"What happens if a listener calls subscribe/unsubscribe while publishing?" — Re-entrancy / borrow issues. Snapshot the listener list before calling, or queue bus mutations, or use channels so dispatch never holds the map borrow across user code.

use std::time::{Duration, Instant};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum State {
    Closed,
    Open,
    HalfOpen,
}

#[derive(Debug, PartialEq)]
enum BreakerError {
    Open,
    CallFailed,
}

struct CircuitBreaker {
    failure_threshold: u32,
    open_duration: Duration,
    state: State,
    consecutive_failures: u32,
    opened_at: Option<Instant>,
}

impl CircuitBreaker {
    fn new(failure_threshold: u32, open_duration: Duration) -> Self {
        Self {
            failure_threshold,
            open_duration,
            state: State::Closed,
            consecutive_failures: 0,
            opened_at: None,
        }
    }

    fn state_at(&mut self, now: Instant) -> State {
        if self.state == State::Open {
            if let Some(opened_at) = self.opened_at {
                if now.duration_since(opened_at) >= self.open_duration {
                    self.state = State::HalfOpen;
                }
            }
        }
        self.state
    }

    fn call<F, T>(&mut self, now: Instant, f: F) -> Result<T, BreakerError>
    where
        F: FnOnce() -> Result<T, ()>,
    {
        match self.state_at(now) {
            State::Open => Err(BreakerError::Open),
            State::Closed | State::HalfOpen => match f() {
                Ok(v) => {
                    self.on_success();
                    Ok(v)
                }
                Err(()) => {
                    self.on_failure(now);
                    Err(BreakerError::CallFailed)
                }
            },
        }
    }

    fn on_success(&mut self) {
        self.consecutive_failures = 0;
        self.state = State::Closed;
        self.opened_at = None;
    }

    fn on_failure(&mut self, now: Instant) {
        match self.state {
            State::HalfOpen => {
                self.state = State::Open;
                self.opened_at = Some(now);
                self.consecutive_failures = self.failure_threshold;
            }
            State::Closed => {
                self.consecutive_failures += 1;
                if self.consecutive_failures >= self.failure_threshold {
                    self.state = State::Open;
                    self.opened_at = Some(now);
                }
            }
            State::Open => {}
        }
    }

    fn state(&self) -> State {
        self.state
    }
}

fn main() {
    let open_for = Duration::from_secs(10);
    let t0 = Instant::now();
    let mut cb = CircuitBreaker::new(2, open_for);

    assert_eq!(
        cb.call(t0, || Err::<i32, ()>(())),
        Err(BreakerError::CallFailed)
    );
    assert_eq!(cb.state(), State::Closed);

    assert_eq!(
        cb.call(t0 + Duration::from_millis(1), || Err::<i32, ()>(())),
        Err(BreakerError::CallFailed)
    );
    assert_eq!(cb.state(), State::Open);

    assert_eq!(
        cb.call(t0 + Duration::from_secs(1), || Ok(42)),
        Err(BreakerError::Open)
    );

    // After cooldown → HalfOpen; success closes.
    // Opened at t0+1ms with 10s duration, so HalfOpen at t0+11s.
    assert_eq!(
        cb.call(t0 + Duration::from_secs(11), || Ok(7)),
        Ok(7)
    );
    assert_eq!(cb.state(), State::Closed);

    // Trip again, then HalfOpen failure re-opens.
    let t1 = t0 + Duration::from_secs(12);
    let _ = cb.call(t1, || Err::<i32, ()>(()));
    let _ = cb.call(t1, || Err::<i32, ()>(()));
    assert_eq!(cb.state(), State::Open);
    assert_eq!(
        cb.call(t1 + Duration::from_secs(10), || Err::<i32, ()>(())),
        Err(BreakerError::CallFailed)
    );
    assert_eq!(cb.state(), State::Open);
}

  • ★★HalfOpen failure re-opens immediately (does not require another full threshold) — matches common breaker semantics and shows they thought about probe failures.
  • ★Clock injection for expiry so open→half-open is assertable without sleeps.

  • P2"Failure rate vs consecutive failures?" — Consecutive count is interview-simple; sliding-window error rates (e.g. 50% of last N) behave better under sparse traffic.
  • P3"Where does this live in a Rust service?" — Around client I/O (HTTP/DB), often as a tower middleware or a small wrapper type storing state beside a connection pool.

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
struct Transfer {
    from: String,
    to: String,
    amount: i64,
}

struct Ledger {
    balances: HashMap<String, i64>,
}

impl Ledger {
    fn new(users: &[&str]) -> Self {
        let mut balances = HashMap::new();
        for u in users {
            balances.insert((*u).to_string(), 0);
        }
        Self { balances }
    }

    fn add_expense(
        &mut self,
        payer: &str,
        amount: i64,
        participants: &[&str],
    ) -> Result<(), &'static str> {
        if amount <= 0 {
            return Err("amount must be positive");
        }
        if participants.is_empty() {
            return Err("no participants");
        }
        if !self.balances.contains_key(payer) {
            return Err("unknown payer");
        }
        for p in participants {
            if !self.balances.contains_key(*p) {
                return Err("unknown participant");
            }
        }

        let n = participants.len() as i64;
        let share = amount / n;
        let mut remainder = amount % n;

        *self.balances.get_mut(payer).unwrap() += amount;

        for p in participants {
            let mut cost = share;
            if remainder > 0 {
                cost += 1;
                remainder -= 1;
            }
            *self.balances.get_mut(*p).unwrap() -= cost;
        }
        Ok(())
    }

    fn balance(&self, user: &str) -> Option<i64> {
        self.balances.get(user).copied()
    }

    /// Greedy settle: repeatedly pair largest debtor with largest creditor.
    fn simplify(&self) -> Vec<Transfer> {
        let mut debtors: Vec<(String, i64)> = Vec::new();
        let mut creditors: Vec<(String, i64)> = Vec::new();

        for (user, &bal) in &self.balances {
            if bal < 0 {
                debtors.push((user.clone(), -bal));
            } else if bal > 0 {
                creditors.push((user.clone(), bal));
            }
        }

        debtors.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
        creditors.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));

        let mut transfers = Vec::new();
        let mut i = 0usize;
        let mut j = 0usize;

        while i < debtors.len() && j < creditors.len() {
            let pay = debtors[i].1.min(creditors[j].1);
            if pay > 0 {
                transfers.push(Transfer {
                    from: debtors[i].0.clone(),
                    to: creditors[j].0.clone(),
                    amount: pay,
                });
            }
            debtors[i].1 -= pay;
            creditors[j].1 -= pay;
            if debtors[i].1 == 0 {
                i += 1;
            }
            if creditors[j].1 == 0 {
                j += 1;
            }
        }
        transfers
    }
}

fn main() {
    let mut ledger = Ledger::new(&["alice", "bob", "carol"]);

    // Alice pays 90 for alice+bob+carol → each owes 30.
    assert!(ledger.add_expense("alice", 90, &["alice", "bob", "carol"]).is_ok());
    assert_eq!(ledger.balance("alice"), Some(60));
    assert_eq!(ledger.balance("bob"), Some(-30));
    assert_eq!(ledger.balance("carol"), Some(-30));

    // Bob pays 30 for bob+carol → each owes 15.
    assert!(ledger.add_expense("bob", 30, &["bob", "carol"]).is_ok());
    assert_eq!(ledger.balance("alice"), Some(60));
    assert_eq!(ledger.balance("bob"), Some(-15));
    assert_eq!(ledger.balance("carol"), Some(-45));

    let mut transfers = ledger.simplify();
    transfers.sort_by(|a, b| {
        a.from
            .cmp(&b.from)
            .then_with(|| a.to.cmp(&b.to))
            .then_with(|| a.amount.cmp(&b.amount))
    });

    // Net: alice +60, bob -15, carol -45 → bob→alice 15, carol→alice 45.
    assert_eq!(
        transfers,
        vec![
            Transfer {
                from: "bob".into(),
                to: "alice".into(),
                amount: 15
            },
            Transfer {
                from: "carol".into(),
                to: "alice".into(),
                amount: 45
            },
        ]
    );

    let total_paid: i64 = transfers.iter().map(|t| t.amount).sum();
    assert_eq!(total_paid, 60);
}
What to look for — unprompted signals
  • ★★★Maintains a single net balance per user (payer credited, participants debited) so simplify does not need the full pairwise IOU matrix.
  • ★★Greedy settle pairs max debtor with max creditor and updates residuals — produces a minimal-ish transfer list good enough for interviews (optimal min-cash-flow is NP-hard).
  • ★Handles integer split remainders explicitly rather than using floats for money.
Follow-up probes
  • P1"Why nets instead of storing every pairwise edge?" — Any expense graph can be reduced to nets without changing who must pay whom in aggregate; simplify then only needs the net vector.
  • P2"Is greedy settle optimal?" — It clears all balances and often uses few transfers, but the minimum number of transactions is a harder problem. Interview bar is correct clearing + clear algorithm.
  • P3"How do you avoid float money bugs?" — Store integer minor units (cents); define remainder policy; never use f64 for balances.
🎯 Senior angle / Rust angle
Expense ledgers reward invariant thinking: sum of balances stays zero after every expense, and simplify is a netting problem. Idiomatic Rust here is HashMap balances, integer cents, and a deterministic sort before greedy pairing so asserts stay stable.
