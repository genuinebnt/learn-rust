use std::collections::{BTreeMap, HashMap, VecDeque};

type Price = i64; // ticks
type Qty = i64;
type OrderId = u64;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Side {
    Bid,
    Ask,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Order {
    id: OrderId,
    side: Side,
    price: Price,
    qty: Qty,
}

struct OrderBook {
    bids: BTreeMap<Price, VecDeque<OrderId>>, // key = -price for max-first iteration
    asks: BTreeMap<Price, VecDeque<OrderId>>,
    orders: HashMap<OrderId, Order>,
    next_id: OrderId,
}

impl OrderBook {
    fn new() -> Self {
        Self {
            bids: BTreeMap::new(),
            asks: BTreeMap::new(),
            orders: HashMap::new(),
            next_id: 1,
        }
    }

    fn add(&mut self, side: Side, price: Price, qty: Qty) -> OrderId {
        assert!(qty > 0);
        let id = self.next_id;
        self.next_id += 1;
        let order = Order {
            id,
            side,
            price,
            qty,
        };
        self.orders.insert(id, order);
        match side {
            Side::Bid => {
                self.bids.entry(-price).or_default().push_back(id);
            }
            Side::Ask => {
                self.asks.entry(price).or_default().push_back(id);
            }
        }
        id
    }

    fn cancel(&mut self, id: OrderId) -> bool {
        let Some(order) = self.orders.remove(&id) else {
            return false;
        };
        let queue = match order.side {
            Side::Bid => self.bids.get_mut(&(-order.price)),
            Side::Ask => self.asks.get_mut(&order.price),
        };
        if let Some(q) = queue {
            if let Some(pos) = q.iter().position(|&x| x == id) {
                q.remove(pos);
            }
            if q.is_empty() {
                match order.side {
                    Side::Bid => {
                        self.bids.remove(&(-order.price));
                    }
                    Side::Ask => {
                        self.asks.remove(&order.price);
                    }
                }
            }
        }
        true
    }

    fn best_bid(&self) -> Option<(Price, Qty)> {
        let (&neg_px, q) = self.bids.iter().next()?;
        let id = *q.front()?;
        let o = self.orders.get(&id)?;
        Some((-neg_px, o.qty))
    }

    fn best_ask(&self) -> Option<(Price, Qty)> {
        let (&px, q) = self.asks.iter().next()?;
        let id = *q.front()?;
        let o = self.orders.get(&id)?;
        Some((px, o.qty))
    }

    /// Aggregate qty at top-of-book price (all FIFO orders at that level).
    fn best_bid_level(&self) -> Option<(Price, Qty)> {
        let (&neg_px, q) = self.bids.iter().next()?;
        let px = -neg_px;
        let mut total = 0;
        for id in q {
            total += self.orders.get(id).map(|o| o.qty).unwrap_or(0);
        }
        Some((px, total))
    }

    fn best_ask_level(&self) -> Option<(Price, Qty)> {
        let (&px, q) = self.asks.iter().next()?;
        let mut total = 0;
        for id in q {
            total += self.orders.get(id).map(|o| o.qty).unwrap_or(0);
        }
        Some((px, total))
    }

    fn order_front_id(&self, side: Side) -> Option<OrderId> {
        match side {
            Side::Bid => self.bids.values().next().and_then(|q| q.front().copied()),
            Side::Ask => self.asks.values().next().and_then(|q| q.front().copied()),
        }
    }
}

fn main() {
    let mut book = OrderBook::new();

    let b1 = book.add(Side::Bid, 100, 5);
    let b2 = book.add(Side::Bid, 100, 3); // same price — behind b1 (time priority)
    let _b3 = book.add(Side::Bid, 99, 10);
    let a1 = book.add(Side::Ask, 101, 4);
    let a2 = book.add(Side::Ask, 102, 7);

    assert_eq!(book.best_bid(), Some((100, 5))); // front of best bid level
    assert_eq!(book.best_bid_level(), Some((100, 8))); // aggregated
    assert_eq!(book.best_ask(), Some((101, 4)));
    assert_eq!(book.best_ask_level(), Some((101, 4)));
    assert_eq!(book.order_front_id(Side::Bid), Some(b1));

    assert!(book.cancel(b1));
    assert_eq!(book.best_bid(), Some((100, 3))); // b2 now front
    assert_eq!(book.order_front_id(Side::Bid), Some(b2));

    assert!(book.cancel(a1));
    assert_eq!(book.best_ask(), Some((102, 7)));
    assert_eq!(book.order_front_id(Side::Ask), Some(a2));

    assert!(!book.cancel(a1)); // already gone
    assert!(book.cancel(b2));
    assert_eq!(book.best_bid_level(), Some((99, 10)));

    println!("tr1 ok");
}
What to look for — unprompted signals
  • ★★★Integer ticks + explicit side queues — never f64 for price; BTreeMap (or sorted vec of levels) with FIFO VecDeque per level.
  • ★★Cancel removes from both id index and level queue and drops empty levels so best bid/ask stay O(log levels) / O(1) front.
  • ★Separates per-order best vs aggregated level qty when asked — shows they understand TOB vs depth.
Follow-up probes
  • P1"BTreeMap vs vec-of-levels?" — BTreeMap is simple and fine for sparse books; dense books often use a contiguous price ladder / indexed levels for cache locality and fewer allocations.
  • P2"Where do allocations hit the hot path?" — HashMap growth, new VecDeque per level, String ids. Pre-size pools; reuse level nodes; avoid format! on every order.
  • P3"How do you encode bid ordering in a single BTreeMap?" — Store bids under -price so ascending iteration yields best bid first, or use two maps / a custom key.

use std::collections::{BTreeMap, HashMap, VecDeque};

type Price = i64;
type Qty = i64;
type OrderId = u64;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Side {
    Buy,
    Sell,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum OrdType {
    Limit,
    Market,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Resting {
    id: OrderId,
    side: Side,
    price: Price,
    qty: Qty,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Trade {
    maker_id: OrderId,
    taker_id: OrderId,
    price: Price,
    qty: Qty,
}

struct Engine {
    bids: BTreeMap<Price, VecDeque<OrderId>>, // keyed by -price
    asks: BTreeMap<Price, VecDeque<OrderId>>,
    orders: HashMap<OrderId, Resting>,
    next_id: OrderId,
    trades: Vec<Trade>,
}

impl Engine {
    fn new() -> Self {
        Self {
            bids: BTreeMap::new(),
            asks: BTreeMap::new(),
            orders: HashMap::new(),
            next_id: 1,
            trades: Vec::new(),
        }
    }

    /// Returns (order_id, unfilled_qty). Limits rest unfilled; markets cancel leftover.
    fn submit(
        &mut self,
        side: Side,
        ord_type: OrdType,
        limit_px: Option<Price>,
        mut qty: Qty,
    ) -> (OrderId, Qty) {
        assert!(qty > 0);
        let taker_id = self.next_id;
        self.next_id += 1;

        while qty > 0 {
            let best = match side {
                Side::Buy => self.best_ask(),
                Side::Sell => self.best_bid(),
            };
            let Some((maker_id, maker_px, maker_qty)) = best else {
                break;
            };

            let crosses = match (side, ord_type, limit_px) {
                (_, OrdType::Market, _) => true,
                (Side::Buy, OrdType::Limit, Some(lp)) => lp >= maker_px,
                (Side::Sell, OrdType::Limit, Some(lp)) => lp <= maker_px,
                _ => false,
            };
            if !crosses {
                break;
            }

            let fill = qty.min(maker_qty);
            self.trades.push(Trade {
                maker_id,
                taker_id,
                price: maker_px,
                qty: fill,
            });
            qty -= fill;
            self.fill_maker(maker_id, fill);
        }

        if qty > 0 {
            if let (OrdType::Limit, Some(px)) = (ord_type, limit_px) {
                self.rest(taker_id, side, px, qty);
            }
        }
        (taker_id, qty)
    }

    fn rest(&mut self, id: OrderId, side: Side, price: Price, qty: Qty) {
        self.orders.insert(
            id,
            Resting {
                id,
                side,
                price,
                qty,
            },
        );
        match side {
            Side::Buy => self.bids.entry(-price).or_default().push_back(id),
            Side::Sell => self.asks.entry(price).or_default().push_back(id),
        }
    }

    fn fill_maker(&mut self, id: OrderId, fill: Qty) {
        let order = self.orders.get_mut(&id).expect("maker");
        order.qty -= fill;
        let side = order.side;
        let price = order.price;
        let remaining = order.qty;
        if remaining == 0 {
            self.orders.remove(&id);
            let empty = match side {
                Side::Buy => {
                    let q = self.bids.get_mut(&(-price)).unwrap();
                    if q.front() == Some(&id) {
                        q.pop_front();
                    } else if let Some(pos) = q.iter().position(|&x| x == id) {
                        q.remove(pos);
                    }
                    q.is_empty()
                }
                Side::Sell => {
                    let q = self.asks.get_mut(&price).unwrap();
                    if q.front() == Some(&id) {
                        q.pop_front();
                    } else if let Some(pos) = q.iter().position(|&x| x == id) {
                        q.remove(pos);
                    }
                    q.is_empty()
                }
            };
            if empty {
                match side {
                    Side::Buy => {
                        self.bids.remove(&(-price));
                    }
                    Side::Sell => {
                        self.asks.remove(&price);
                    }
                }
            }
        }
    }

    fn best_ask(&self) -> Option<(OrderId, Price, Qty)> {
        let (&px, q) = self.asks.iter().next()?;
        let id = *q.front()?;
        let o = self.orders.get(&id)?;
        Some((id, px, o.qty))
    }

    fn best_bid(&self) -> Option<(OrderId, Price, Qty)> {
        let (&neg, q) = self.bids.iter().next()?;
        let id = *q.front()?;
        let o = self.orders.get(&id)?;
        Some((id, -neg, o.qty))
    }
}

fn main() {
    let mut eng = Engine::new();

    let (sell1, rest1) = eng.submit(Side::Sell, OrdType::Limit, Some(100), 10);
    assert_eq!(rest1, 10);
    let (sell2, rest2) = eng.submit(Side::Sell, OrdType::Limit, Some(101), 5);
    assert_eq!(rest2, 5);

    let (buy1, resting_buy) = eng.submit(Side::Buy, OrdType::Limit, Some(101), 12);
    assert_eq!(resting_buy, 0);
    assert_eq!(eng.trades.len(), 2);
    assert_eq!(
        eng.trades[0],
        Trade {
            maker_id: sell1,
            taker_id: buy1,
            price: 100,
            qty: 10
        }
    );
    assert_eq!(
        eng.trades[1],
        Trade {
            maker_id: sell2,
            taker_id: buy1,
            price: 101,
            qty: 2
        }
    );
    assert_eq!(eng.best_ask().map(|(_, p, q)| (p, q)), Some((101, 3)));

    // Non-crossing limit rests
    let (bid_id, left) = eng.submit(Side::Buy, OrdType::Limit, Some(99), 4);
    assert_eq!(left, 4);
    assert_eq!(eng.best_bid().map(|(id, p, q)| (id, p, q)), Some((bid_id, 99, 4)));

    // Market sell takes the bid; leftover cancelled (not rested)
    let (mkt, unfilled) = eng.submit(Side::Sell, OrdType::Market, None, 10);
    assert_eq!(unfilled, 6);
    assert_eq!(
        eng.trades.last().cloned(),
        Some(Trade {
            maker_id: bid_id,
            taker_id: mkt,
            price: 99,
            qty: 4
        })
    );
    assert!(eng.best_bid().is_none());

    // Limit that does not cross leaves book unchanged on ask side
    let before = eng.trades.len();
    let (_id, rem) = eng.submit(Side::Buy, OrdType::Limit, Some(100), 1);
    assert_eq!(rem, 1); // best ask is 101; does not cross
    assert_eq!(eng.trades.len(), before);

    println!("tr2 ok");
}

use std::sync::mpsc::{self, Receiver, Sender};

type Price = i64;
type Qty = i64;

#[derive(Debug, Clone, PartialEq, Eq)]
struct BookTop {
    bid: Option<(Price, Qty)>,
    ask: Option<(Price, Qty)>,
    seq: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum MdEvent {
    Snapshot(BookTop),
    Delta {
        seq: u64,
        bid: Option<(Price, Qty)>,
        ask: Option<(Price, Qty)>,
    },
}

struct Fanout {
    seq: u64,
    top: BookTop,
    subs: Vec<(u64, Sender<MdEvent>)>,
    next_sub: u64,
}

impl Fanout {
    fn new() -> Self {
        Self {
            seq: 0,
            top: BookTop {
                bid: None,
                ask: None,
                seq: 0,
            },
            subs: Vec::new(),
            next_sub: 1,
        }
    }

    fn subscribe(&mut self) -> (u64, Receiver<MdEvent>) {
        let (tx, rx) = mpsc::channel();
        let id = self.next_sub;
        self.next_sub += 1;
        // Snapshot first so late joiners catch up
        let _ = tx.send(MdEvent::Snapshot(self.top.clone()));
        self.subs.push((id, tx));
        (id, rx)
    }

    fn unsubscribe(&mut self, id: u64) {
        self.subs.retain(|(sid, _)| *sid != id);
    }

    fn publish_top(&mut self, bid: Option<(Price, Qty)>, ask: Option<(Price, Qty)>) {
        self.seq += 1;
        self.top = BookTop {
            bid,
            ask,
            seq: self.seq,
        };
        let ev = MdEvent::Delta {
            seq: self.seq,
            bid,
            ask,
        };
        self.subs.retain(|(_, tx)| tx.send(ev.clone()).is_ok());
    }
}

fn main() {
    let mut bus = Fanout::new();
    bus.publish_top(Some((100, 5)), Some((101, 3)));

    let (a_id, a_rx) = bus.subscribe();
    let (_b_id, b_rx) = bus.subscribe();

    // Both get snapshot of current top
    match a_rx.recv().unwrap() {
        MdEvent::Snapshot(s) => {
            assert_eq!(s.bid, Some((100, 5)));
            assert_eq!(s.ask, Some((101, 3)));
            assert_eq!(s.seq, 1);
        }
        other => panic!("expected snapshot, got {:?}", other),
    }
    match b_rx.recv().unwrap() {
        MdEvent::Snapshot(s) => assert_eq!(s.seq, 1),
        other => panic!("expected snapshot, got {:?}", other),
    }

    bus.publish_top(Some((100, 8)), Some((101, 2)));

    match a_rx.recv().unwrap() {
        MdEvent::Delta { seq, bid, ask } => {
            assert_eq!(seq, 2);
            assert_eq!(bid, Some((100, 8)));
            assert_eq!(ask, Some((101, 2)));
        }
        other => panic!("expected delta, got {:?}", other),
    }
    match b_rx.recv().unwrap() {
        MdEvent::Delta { seq, .. } => assert_eq!(seq, 2),
        other => panic!("expected delta, got {:?}", other),
    }

    bus.unsubscribe(a_id);
    bus.publish_top(Some((99, 1)), Some((101, 2)));
    // a is gone — must not block; b still receives
    assert!(a_rx.try_recv().is_err());
    match b_rx.recv().unwrap() {
        MdEvent::Delta { seq, bid, .. } => {
            assert_eq!(seq, 3);
            assert_eq!(bid, Some((99, 1)));
        }
        other => panic!("expected delta, got {:?}", other),
    }

    println!("tr3 ok");
}
What to look for — unprompted signals
  • ★★★Snapshot-on-subscribe + sequenced deltas — late joiners are consistent without replaying the world.
  • ★★Prunes failed sends / unsubscribe by id so one slow or dropped consumer does not poison the bus.
  • ★Clone-small events (Copy tops or Arc payload) and can discuss broadcast vs per-sub mpsc.
Follow-up probes
  • P1"mpsc vs broadcast?" — Per-sub mpsc gives independent backpressure; broadcast shares one stream with lag. Pick based on whether consumers may stall.
  • P2"What if a subscriber never drains?" — Bounded channels + drop-oldest / disconnect policy; never let MD publish block the matching thread.
  • P3"Snapshot vs incremental recovery after gap?" — Detect seq gap → request snapshot; deltas alone are not enough after loss.

type Qty = i64; // signed: +long / -short
type Price = i64; // ticks
type Notional = i64; // price * qty in tick*share units

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Side {
    Buy,
    Sell,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum RiskReject {
    PositionLimit { projected: Qty, limit: Qty },
    NotionalLimit { projected: Notional, limit: Notional },
    BadOrder,
}

#[derive(Debug, Clone)]
struct RiskLimits {
    /// Absolute max |position|.
    max_abs_position: Qty,
    /// Max |gross notional| of open exposure after the order (mark * position).
    max_abs_notional: Notional,
}

struct RiskGateway {
    limits: RiskLimits,
    position: Qty,
    /// Last mark used for notional (integer ticks).
    mark: Price,
}

impl RiskGateway {
    fn new(limits: RiskLimits, mark: Price) -> Self {
        Self {
            limits,
            position: 0,
            mark,
        }
    }

    fn set_mark(&mut self, mark: Price) {
        self.mark = mark;
    }

    fn check(&self, side: Side, qty: Qty, limit_px: Price) -> Result<(), RiskReject> {
        if qty <= 0 || limit_px <= 0 {
            return Err(RiskReject::BadOrder);
        }
        let signed = match side {
            Side::Buy => qty,
            Side::Sell => -qty,
        };
        let projected = self.position + signed;
        if projected.abs() > self.limits.max_abs_position {
            return Err(RiskReject::PositionLimit {
                projected,
                limit: self.limits.max_abs_position,
            });
        }
        // Conservative: use max(mark, limit_px) for buys / min for sells? Demo: use order px.
        let projected_notional = projected.abs().saturating_mul(limit_px);
        if projected_notional > self.limits.max_abs_notional {
            return Err(RiskReject::NotionalLimit {
                projected: projected_notional,
                limit: self.limits.max_abs_notional,
            });
        }
        let _ = self.mark; // available for mark-based checks in probes
        Ok(())
    }

    /// Apply fill after matching (only call when check passed / fill confirmed).
    fn on_fill(&mut self, side: Side, qty: Qty) {
        match side {
            Side::Buy => self.position += qty,
            Side::Sell => self.position -= qty,
        }
    }

    fn try_order(&mut self, side: Side, qty: Qty, limit_px: Price) -> Result<(), RiskReject> {
        self.check(side, qty, limit_px)?;
        // In a real gateway, order is forwarded; here we simulate immediate full fill.
        self.on_fill(side, qty);
        Ok(())
    }
}

fn main() {
    let limits = RiskLimits {
        max_abs_position: 100,
        max_abs_notional: 50_000, // e.g. 100 * 500
    };
    let mut gw = RiskGateway::new(limits, 400);
    gw.set_mark(410);

const SOH: char = '\u{1}';

#[derive(Debug, Clone, PartialEq, Eq)]
enum MsgType {
    HeartBeat,      // 0
    NewOrderSingle, // D

#[derive(Debug, Clone, PartialEq, Eq)]
struct FixMessage {
    msg_type: MsgType,
    seq: u64,
    /// Present for NewOrderSingle
    cl_ord_id: Option<String>,
    symbol: Option<String>,
    side: Option<char>,  // '1' buy / '2' sell
    qty: Option<i64>,    // tag 38
    price: Option<i64>,  // tag 44, integer ticks
}

impl FixMessage {
    fn heartbeat(seq: u64) -> Self {
        Self {
            msg_type: MsgType::HeartBeat,
            seq,
            cl_ord_id: None,
            symbol: None,
            side: None,
            qty: None,
            price: None,
        }
    }

    fn new_order(
        seq: u64,
        cl_ord_id: &str,
        symbol: &str,
        side: char,
        qty: i64,
        price: i64,
    ) -> Self {
        Self {
            msg_type: MsgType::NewOrderSingle,
            seq,
            cl_ord_id: Some(cl_ord_id.into()),
            symbol: Some(symbol.into()),
            side: Some(side),
            qty: Some(qty),
            price: Some(price),
        }
    }

    fn serialize(&self) -> String {
        let mut fields: Vec<(u32, String)> = Vec::new();
        // 35=MsgType, 34=MsgSeqNum, then body
        let mt = match self.msg_type {
            MsgType::HeartBeat => "0",
            MsgType::NewOrderSingle => "D",
        };
        fields.push((35, mt.into()));
        fields.push((34, self.seq.to_string()));
        if let Some(ref c) = self.cl_ord_id {
            fields.push((11, c.clone()));
        }
        if let Some(ref s) = self.symbol {
            fields.push((55, s.clone()));
        }
        if let Some(side) = self.side {
            fields.push((54, side.to_string()));
        }
        if let Some(q) = self.qty {
            fields.push((38, q.to_string()));
        }
        if let Some(p) = self.price {
            fields.push((44, p.to_string()));
        }
        // BodyLength (9) over everything after 9=…| — simplified: length of body from 35 onward
        let body: String = fields
            .iter()
            .map(|(t, v)| format!("{}={}{}", t, v, SOH))
            .collect();
        let body_len = body.len();
        let mut out = format!("8=FIX.4.2{}9={}{}", SOH, body_len, SOH);
        out.push_str(&body);
        // Checksum tag 10 = sum of bytes mod 256 (excluding the checksum field itself)
        let sum: u32 = out.bytes().map(|b| b as u32).sum();
        out.push_str(&format!("10={:03}{}", sum % 256, SOH));
        out
    }

    fn parse(raw: &str) -> Result<Self, String> {
        let mut map = std::collections::HashMap::new();
        for part in raw.split(SOH) {
            if part.is_empty() {
                continue;
            }
            let (tag, val) = part
                .split_once('=')
                .ok_or_else(|| format!("bad field: {}", part))?;
            let tag: u32 = tag.parse().map_err(|_| format!("bad tag: {}", tag))?;
            map.insert(tag, val.to_string());
        }
        let mt = map.get(&35).ok_or("missing 35")?;
        let msg_type = match mt.as_str() {
            "0" => MsgType::HeartBeat,
            "D" => MsgType::NewOrderSingle,
            other => return Err(format!("unsupported MsgType {}", other)),
        };
        let seq: u64 = map
            .get(&34)
            .ok_or("missing 34")?
            .parse()
            .map_err(|_| "bad 34")?;
        let parse_i64 = |t: u32| -> Option<i64> { map.get(&t).and_then(|s| s.parse().ok()) };
        Ok(FixMessage {
            msg_type,
            seq,
            cl_ord_id: map.get(&11).cloned(),
            symbol: map.get(&55).cloned(),
            side: map.get(&54).and_then(|s| s.chars().next()),
            qty: parse_i64(38),
            price: parse_i64(44),
        })
    }
}

type Qty = i64;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Side {
    Buy,
    Sell,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct ChildOrder {
    slice_idx: usize,
    release_at_ms: u64,
    side: Side,
    qty: Qty,
}

struct TwapScheduler {
    side: Side,
    parent_qty: Qty,
    start_ms: u64,
    end_ms: u64,
    n_slices: usize,
    children: Vec<ChildOrder>,
    next_idx: usize,
}

impl TwapScheduler {
    fn new(side: Side, parent_qty: Qty, start_ms: u64, end_ms: u64, n_slices: usize) -> Self {
        assert!(parent_qty > 0);
        assert!(n_slices > 0);
        assert!(end_ms > start_ms);
        let base = parent_qty / n_slices as Qty;
        let mut rem = parent_qty % n_slices as Qty;
        let span = end_ms - start_ms;
        let mut children = Vec::with_capacity(n_slices);
        for i in 0..n_slices {
            let mut q = base;
            if rem > 0 {
                q += 1;
                rem -= 1;
            }
            // Release at start of each slice; last slice still before end.
            let release_at_ms = start_ms + (span * i as u64) / n_slices as u64;
            children.push(ChildOrder {
                slice_idx: i,
                release_at_ms,
                side,
                qty: q,
            });
        }
        // Remainder distributed to earliest slices → sum == parent
        debug_assert_eq!(children.iter().map(|c| c.qty).sum::<Qty>(), parent_qty);
        Self {
            side,
            parent_qty,
            start_ms,
            end_ms,
            n_slices,
            children,
            next_idx: 0,
        }
    }

    /// Poll due children at `now_ms` (inject clock — no Instant::now in lib).
    fn poll(&mut self, now_ms: u64) -> Vec<ChildOrder> {
        let mut due = Vec::new();
        while self.next_idx < self.children.len() {
            if self.children[self.next_idx].release_at_ms <= now_ms {
                due.push(self.children[self.next_idx].clone());
                self.next_idx += 1;
            } else {
                break;
            }
        }
        due
    }

    fn remaining_parent_qty(&self) -> Qty {
        self.children[self.next_idx..]
            .iter()
            .map(|c| c.qty)
            .sum()
    }
}

fn main() {
    let mut twap = TwapScheduler::new(Side::Buy, 100, 1_000, 5_000, 4);
    // slices of 25 each; releases at 1000, 2000, 3000, 4000
    assert_eq!(twap.children.len(), 4);
    assert_eq!(twap.children.iter().map(|c| c.qty).sum::<Qty>(), 100);
    assert_eq!(twap.children[0].release_at_ms, 1_000);
    assert_eq!(twap.children[1].release_at_ms, 2_000);
    assert_eq!(twap.children[2].release_at_ms, 3_000);
    assert_eq!(twap.children[3].release_at_ms, 4_000);

    assert!(twap.poll(999).is_empty());
    let d0 = twap.poll(1_000);
    assert_eq!(d0.len(), 1);
    assert_eq!(d0[0].qty, 25);

    let d1 = twap.poll(3_500); // two more due
    assert_eq!(d1.len(), 2);
    assert_eq!(d1[0].slice_idx, 1);

    assert_eq!(twap.remaining_parent_qty(), 0);

type Qty = i64;
type Price = i64; // ticks
type Money = i64; // tick * qty (PnL in tick-shares)

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Side {
    Buy,
    Sell,
}

struct PositionTracker {
    qty: Qty,
    /// Average entry price in ticks for the open position (valid when qty != 0).
    avg_px: Price,
    realized: Money,
}

impl PositionTracker {
    fn new() -> Self {
        Self {
            qty: 0,
            avg_px: 0,
            realized: 0,
        }
    }

    fn on_fill(&mut self, side: Side, fill_qty: Qty, fill_px: Price) {
        assert!(fill_qty > 0);
        let signed = match side {
            Side::Buy => fill_qty,
            Side::Sell => -fill_qty,
        };

        if self.qty == 0 || self.qty.signum() == signed.signum() {
            // Increasing / opening same direction — VWAP average
            let new_qty = self.qty + signed;
            let abs_old = self.qty.abs();
            let abs_new = new_qty.abs();
            let abs_fill = fill_qty;
            self.avg_px = if abs_new == 0 {
                0
            } else {
                (self.avg_px * abs_old + fill_px * abs_fill) / abs_new
            };
            self.qty = new_qty;
            return;
        }

        // Reducing / flipping
        let closing = fill_qty.min(self.qty.abs());
        let direction = self.qty.signum(); // +1 long, -1 short
        // Realized: long closes on sell → (sell_px - avg) * qty; short closes on buy → (avg - buy_px) * qty
        let pnl_per = if direction > 0 {
            fill_px - self.avg_px
        } else {
            self.avg_px - fill_px
        };
        self.realized += pnl_per * closing;

        let new_qty = self.qty + signed;
        if new_qty == 0 {
            self.qty = 0;
            self.avg_px = 0;
        } else if new_qty.signum() == direction {
            // Partial reduce — avg unchanged
            self.qty = new_qty;
        } else {
            // Flip: remainder opens at fill_px
            self.qty = new_qty;
            self.avg_px = fill_px;
        }
    }

    fn unrealized(&self, mark: Price) -> Money {
        if self.qty == 0 {
            return 0;
        }
        if self.qty > 0 {
            (mark - self.avg_px) * self.qty
        } else {
            (self.avg_px - mark) * (-self.qty)
        }
    }

    fn total_pnl(&self, mark: Price) -> Money {
        self.realized + self.unrealized(mark)
    }
}

fn main() {
    let mut p = PositionTracker::new();

    // Buy 10 @ 100
    p.on_fill(Side::Buy, 10, 100);
    assert_eq!(p.qty, 10);
    assert_eq!(p.avg_px, 100);
    assert_eq!(p.unrealized(110), 100); // 10 * 10
    assert_eq!(p.realized, 0);

    // Buy 10 @ 120 → avg 110
    p.on_fill(Side::Buy, 10, 120);
    assert_eq!(p.qty, 20);
    assert_eq!(p.avg_px, 110);

    // Sell 5 @ 130 → realize (130-110)*5 = 100
    p.on_fill(Side::Sell, 5, 130);
    assert_eq!(p.qty, 15);
    assert_eq!(p.avg_px, 110);
    assert_eq!(p.realized, 100);
    assert_eq!(p.unrealized(110), 0);
    assert_eq!(p.total_pnl(110), 100);

    // Sell 20 @ 100 — close 15 realizing (100-110)*15 = -150, flip short 5 @ 100
    p.on_fill(Side::Sell, 20, 100);
    assert_eq!(p.qty, -5);
    assert_eq!(p.avg_px, 100);
    assert_eq!(p.realized, 100 - 150); // -50
    // Short 5 @ avg 100, mark 90 → unrealized +50
    assert_eq!(p.unrealized(90), 50);
    assert_eq!(p.total_pnl(90), 0);

    // Cover short
    p.on_fill(Side::Buy, 5, 90);
    assert_eq!(p.qty, 0);
    assert_eq!(p.realized, -50 + 50); // 0
    assert_eq!(p.unrealized(999), 0);

    ask: i64,
}

struct SpscRing<T> {
    buf: Box<[UnsafeCell<Option<T>>]>,
    capacity: usize,
    /// Next write index (producer).
    head: AtomicUsize,
    /// Next read index (consumer).
    tail: AtomicUsize,
}

unsafe impl<T: Send> Send for SpscRing<T> {}
unsafe impl<T: Send> Sync for SpscRing<T> {}

impl<T> SpscRing<T> {
    fn with_capacity(capacity: usize) -> Self {
        assert!(capacity.is_power_of_two());
        let mut v = Vec::with_capacity(capacity);
        for _ in 0..capacity {
            v.push(UnsafeCell::new(None));
        }
        Self {
            buf: v.into_boxed_slice(),
            capacity,
            head: AtomicUsize::new(0),
            tail: AtomicUsize::new(0),
        }
    }

    fn push(&self, item: T) -> Result<(), T> {
        let head = self.head.load(Ordering::Relaxed);
        let tail = self.tail.load(Ordering::Acquire);
        if head.wrapping_sub(tail) == self.capacity {
            return Err(item); // full
        }
        let idx = head & (self.capacity - 1);
        unsafe {
            *self.buf[idx].get() = Some(item);
        }
        self.head.store(head.wrapping_add(1), Ordering::Release);
        Ok(())
    }

    fn pop(&self) -> Option<T> {
        let tail = self.tail.load(Ordering::Relaxed);
        let head = self.head.load(Ordering::Acquire);
        if tail == head {
            return None; // empty
        }
        let idx = tail & (self.capacity - 1);
        let item = unsafe { (*self.buf[idx].get()).take() };
        self.tail.store(tail.wrapping_add(1), Ordering::Release);
        item
    }
}

fn main() {
    // Single-thread correctness
    let ring = SpscRing::with_capacity(4);
    assert!(ring.pop().is_none());
    assert!(ring
        .push(Tick {
            seq: 1,
            bid: 100,
            ask: 101
        })
        .is_ok());
    assert!(ring
        .push(Tick {
            seq: 2,
            bid: 100,
            ask: 102
        })
        .is_ok());
    assert_eq!(
        ring.pop(),
        Some(Tick {
            seq: 1,
            bid: 100,
            ask: 101
        })
    );
    assert_eq!(ring.pop().map(|t| t.seq), Some(2));
    assert!(ring.pop().is_none());

    // Fill to capacity
    for i in 0..4 {
        assert!(ring
            .push(Tick {
                seq: 10 + i,
                bid: 1,
                ask: 2
            })
            .is_ok());
    }
    assert!(ring
        .push(Tick {
            seq: 99,
            bid: 0,
            ask: 0
        })
        .is_err());
    assert_eq!(ring.pop().map(|t| t.seq), Some(10));
    assert!(ring
        .push(Tick {
            seq: 14,
            bid: 1,
            ask: 2
        })
        .is_ok());

    // Cross-thread SPSC
    let shared = Arc::new(SpscRing::with_capacity(1024));
    let prod = Arc::clone(&shared);
    let cons = Arc::clone(&shared);
    let n = 50_000u64;
    let h = thread::spawn(move || {
        for i in 0..n {
            let tick = Tick {
                seq: i,
                bid: i as i64,
                ask: i as i64 + 1,
            };
            while prod.push(tick).is_err() {
                std::hint::spin_loop();
            }
        }
    });
    let mut got = 0u64;
    let mut last = None;
    while got < n {
        if let Some(t) = cons.pop() {
            assert_eq!(t.seq, got);
            last = Some(t);
            got += 1;
        } else {
            std::hint::spin_loop();
        }
    }
    h.join().unwrap();
    assert_eq!(
        last,
        Some(Tick {
            seq: n - 1,
            bid: (n - 1) as i64,
            ask: n as i64
        })
    );

    println!("tr8 ok");
}
What to look for — unprompted signals
  • ★★★Correct Acquire/Release pairing on head/tail so the payload write happens-before the index publish.
  • ★★Power-of-two mask indexing and full/empty via head - tail == cap / equality — no locks on the happy path.
  • ★UnsafeCell + Send/Sync rationale articulated — single producer / single consumer is what makes races impossible by construction.
Follow-up probes
  • P1"Why SPSC not MPSC here?" — One matching/publisher thread → one consumer ( nic / encoder ) is the common MD path; MPSC needs claim slots or a mutex.
  • P2"What Ordering is required?" — Release on publish of index, Acquire on load of the opposite index; Relaxed on the local-owned cursor load.
  • P3"Cache-line padding?" — False sharing on adjacent head/tail atomics; pad to 64B in production rings (crossbeam, custom hot path).
